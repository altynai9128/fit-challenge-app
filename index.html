function getGroupMembers(groupId) { const sheet = getSheet('members'); const data = sheet.getDataRange().getValues(); const members = []; for (let i = 1; i < data.length; i++) { if (data[i][0] == groupId) { // Ищем пользователя по ID const userSheet = getSheet('users'); const userData = userSheet.getDataRange().getValues(); let photoUrl = ''; let name = data[i][2] || ''; for (let j = 1; j < userData.length; j++) { if (userData[j][0] == data[i][1]) { photoUrl = userData[j][11] || ''; break; } } members.push({ id: data[i][1], name: data[i][2] || 'Участник', paid: data[i][3] || false, startWeight: data[i][4] || '', currentWeight: data[i][5] || '', weightHistory: data[i][6] || '', photoBefore: data[i][7] || '', photoAfter: data[i][8] || '', photoUrl: photoUrl }); } } return { status: 'ok', members: members }; }
